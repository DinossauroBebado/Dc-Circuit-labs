5. Código Fonte (main.cpp)

C++ — TENSAO_VIN/main.cpp
#include <Arduino.h>

#define PIN_TENSAO_VIN 1
#define NUM_AMOSTRAS 16
#define ESCALA_MAX_VIN 45.0f

int lerADCComMedia(uint8_t pin)
{
    long soma = 0;
    for (int i = 0; i < NUM_AMOSTRAS; i++)
    {
        soma += analogRead(pin);
        delayMicroseconds(100);
    }
    return (int)(soma / NUM_AMOSTRAS);
}

void setup()
{
    Serial.begin(115200);
    delay(2000);
    analogReadResolution(12);
    analogSetAttenuation(ADC_11db);
    pinMode(PIN_TENSAO_VIN, INPUT);
    gpio_pullup_dis((gpio_num_t)PIN_TENSAO_VIN);
    gpio_pulldown_dis((gpio_num_t)PIN_TENSAO_VIN);

    Serial.println("=================================================");
    Serial.println(" DC Circuit Labs - Leitura Tensão Entrada (V_IN)");
    Serial.println("=================================================");
    Serial.printf("Pino: GPIO%d (ADC1_CH0)\n", PIN_TENSAO_VIN);
    Serial.printf("Resolução: 12 bits | Faixa: 0.0V a %.1fV DC\n\n", ESCALA_MAX_VIN);
}

void loop()
{
    int adcRaw = lerADCComMedia(PIN_TENSAO_VIN);
    float tensaoPinoESP = (adcRaw / 4095.0f) * 3.3f;
    float tensaoEntradaVin = (adcRaw / 4095.0f) * ESCALA_MAX_VIN;

    Serial.println("-------------------------------------------------");
    Serial.printf("Leitura ADC Bruta (0-4095)    : %4d\n", adcRaw);
    Serial.printf("Tensão no Pino ESP32 (0-3.3V) : %.2f V\n", tensaoPinoESP);
    Serial.printf("Tensão de Entrada Placa V_IN  : %.2f V\n", tensaoEntradaVin);
    Serial.println("-------------------------------------------------\n");
    delay(1000);
}

5. Código Fonte (main.cpp)

C++ — RTC/main.cpp
#include <Arduino.h>
#include <Wire.h>
#include <RTClib.h>

RTC_PCF8563 rtc;

void setup()
{
    Serial.begin(115200);
    delay(2000);
    Serial.println("DC Circuit Labs - Iniciando Teste do RTC PCF8563...");

    Wire.begin();
    if (!rtc.begin())
    {
        Serial.println("ERRO: Chip PCF8563 não encontrado!");
        Serial.flush();
        while (1) delay(10);
    }

    if (rtc.lostPower())
    {
        Serial.println("Aviso: RTC perdeu energia. Ajustando para hora da compilação!");
        rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    }
    Serial.println("Hardware validado. PCF8563 Online.");
}

void loop()
{
    DateTime now = rtc.now();
    Serial.print("Data/Hora Atual: ");
    Serial.printf("%04d/%02d/%02d %02d:%02d:%02d\n",
        now.year(), now.month(), now.day(),
        now.hour(), now.minute(), now.second());
    delay(1000);
}

4. Código Fonte (main.cpp)

C++ — RELES/main.cpp
#include <Arduino.h>

#define PIN_LATCH 21
#define PIN_CLOCK 12
#define PIN_DATA 11
#define TOTAL_RELES 7

void atualizarReles(uint8_t estadoReles)
{
    digitalWrite(PIN_LATCH, LOW);
    shiftOut(PIN_DATA, PIN_CLOCK, MSBFIRST, estadoReles);
    digitalWrite(PIN_LATCH, HIGH);
}

void setup()
{
    Serial.begin(115200);
    delay(2000);
    pinMode(PIN_LATCH, OUTPUT);
    pinMode(PIN_CLOCK, OUTPUT);
    pinMode(PIN_DATA, OUTPUT);
    atualizarReles(0x00);

    Serial.println("=================================================");
    Serial.println(" DC Circuit Labs - Teste de Acionamento dos Relés");
    Serial.println("=================================================");
}

void loop()
{
    Serial.println(">>> Sequência Avanço (Relé 1 -> Relé 7)...");
    for (int i = 0; i < TOTAL_RELES; i++)
    {
        uint8_t mascara = (1 << i);
        atualizarReles(mascara);
        Serial.printf("  [LIGADO] Relé %d\n", i + 1);
        delay(1000);
    }

    Serial.println("\n>>> Sequência Retorno (Relé 6 -> Relé 2)...");
    for (int i = TOTAL_RELES - 2; i > 0; i--)
    {
        uint8_t mascara = (1 << i);
        atualizarReles(mascara);
        Serial.printf("  [LIGADO] Relé %d\n", i + 1);
        delay(1000);
    }

    Serial.println("\n>>> Desligando todos os relés por 1s...\n");
    atualizarReles(0x00);
    delay(1000);
}

5. Código Fonte (main.cpp)

C++ — BUZZER/main.cpp
#include <Arduino.h>

#define PIN_LATCH 21
#define PIN_CLOCK 12
#define PIN_DATA  11
#define MASK_BUZZER (1 << 7)

#define NOTE_G4  392
#define NOTE_DS4 311
#define NOTE_AS4 466
#define NOTE_D5  587
#define NOTE_DS5 622
#define NOTE_FS4 370
#define NOTE_G5  784
#define NOTE_F5  698

void atualizarSaida(uint8_t valorByte)
{
    digitalWrite(PIN_LATCH, LOW);
    shiftOut(PIN_DATA, PIN_CLOCK, MSBFIRST, valorByte);
    digitalWrite(PIN_LATCH, HIGH);
}

void setBuzzer(bool ligado)
{
    atualizarSaida(ligado ? MASK_BUZZER : 0x00);
}

void tocarNota(int frequenciaHz, int duracaoMs)
{
    if (frequenciaHz == 0) { setBuzzer(false); delay(duracaoMs); return; }
    long meioPeriodoMicros = 1000000L / (frequenciaHz * 2);
    long ciclos = ((long)duracaoMs * 1000L) / (meioPeriodoMicros * 2);
    for (long i = 0; i < ciclos; i++)
    {
        setBuzzer(true);  delayMicroseconds(meioPeriodoMicros);
        setBuzzer(false); delayMicroseconds(meioPeriodoMicros);
    }
    delay(50);
}

void setup()
{
    Serial.begin(115200); delay(2000);
    pinMode(PIN_LATCH, OUTPUT);
    pinMode(PIN_CLOCK, OUTPUT);
    pinMode(PIN_DATA, OUTPUT);
    setBuzzer(false);
    Serial.println("DC Circuit Labs - Melodia: Marcha Imperial");
}

void loop()
{
    // Frase 1
    tocarNota(NOTE_G4, 500); tocarNota(NOTE_G4, 500); tocarNota(NOTE_G4, 500);
    tocarNota(NOTE_DS4, 350); tocarNota(NOTE_AS4, 150);
    tocarNota(NOTE_G4, 500); tocarNota(NOTE_DS4, 350);
    tocarNota(NOTE_AS4, 150); tocarNota(NOTE_G4, 1000);
    // Frase 2
    tocarNota(NOTE_D5, 500); tocarNota(NOTE_D5, 500); tocarNota(NOTE_D5, 500);
    tocarNota(NOTE_DS5, 350); tocarNota(NOTE_AS4, 150);
    tocarNota(NOTE_FS4, 500); tocarNota(NOTE_DS4, 350);
    tocarNota(NOTE_AS4, 150); tocarNota(NOTE_G4, 1000);
    // Frase 3
    tocarNota(NOTE_G5, 500); tocarNota(NOTE_G4, 350); tocarNota(NOTE_G4, 150);
    tocarNota(NOTE_G5, 500); tocarNota(NOTE_FS4, 350); tocarNota(NOTE_F5, 150);
    tocarNota(NOTE_DS4, 350); tocarNota(NOTE_AS4, 150); tocarNota(NOTE_G4, 800);
    delay(3000);
}

4. Código Fonte (main.cpp)

C++ — COMUNICACAO_IHM/main.cpp
#include <Arduino.h>

// RS485 (THVD1450DR)
#define PIN_RS485_R     41
#define PIN_RS485_RE_DE 40
#define PIN_RS485_D     42
#define RS485_BAUD_RATE 9600

// Relés (74HC595D + ULN2803)
#define PIN_LATCH 21
#define PIN_CLOCK 12
#define PIN_DATA  11

uint8_t estadoReles = 0x00;
unsigned long ultimoEnvio = 0;
const unsigned long INTERVALO_ENVIO = 20000;
unsigned long numeroSequencial = 1;

void atualizarReles(uint8_t estado)
{
    digitalWrite(PIN_LATCH, LOW);
    shiftOut(PIN_DATA, PIN_CLOCK, MSBFIRST, estado);
    digitalWrite(PIN_LATCH, HIGH);
}

void enviarNumeroRS485()
{
    digitalWrite(PIN_RS485_RE_DE, HIGH);
    delay(5);
    String pacote = "NUM:" + String(numeroSequencial) + "\n";
    Serial1.print(pacote);
    Serial1.flush();
    delay(5);
    digitalWrite(PIN_RS485_RE_DE, LOW);
    Serial.printf("[TX] Número enviado: %lu\n", numeroSequencial);
    numeroSequencial++;
}

void setup()
{
    Serial.begin(115200); delay(2000);
    Serial.println("=================================================");
    Serial.println(" DC Circuit Labs - Placa Rogue v4.1 (IHM Test)  ");
    Serial.println("=================================================");

    pinMode(PIN_LATCH, OUTPUT);
    pinMode(PIN_CLOCK, OUTPUT);
    pinMode(PIN_DATA, OUTPUT);
    atualizarReles(estadoReles);

    pinMode(PIN_RS485_RE_DE, OUTPUT);
    digitalWrite(PIN_RS485_RE_DE, LOW);
    Serial1.begin(RS485_BAUD_RATE, SERIAL_8N1, PIN_RS485_R, PIN_RS485_D);

    enviarNumeroRS485();
    ultimoEnvio = millis();
}

void loop()
{
    if (millis() - ultimoEnvio >= INTERVALO_ENVIO)
    {
        ultimoEnvio = millis();
        enviarNumeroRS485();
    }

    if (Serial1.available())
    {
        String cmd = Serial1.readStringUntil('\n');
        cmd.trim();
        if (cmd.length() > 0)
        {
            Serial.printf("[RX] Comando: \"%s\"\n", cmd.c_str());
            if (cmd == "CMD:RELE1_TOGGLE" || cmd == "CMD:RELE1")
            {
                estadoReles ^= (1 << 0);
                atualizarReles(estadoReles);
                Serial.printf("  => Relé 1 %s! (0x%02X)\n",
                    (estadoReles & 0x01) ? "LIGADO" : "DESLIGADO", estadoReles);
            }
        }
    }
    delay(10);
}
